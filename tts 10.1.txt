async function callAI(system, user) {
    const apiKey = "";
    const url = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-09-2025:generateContent?key=${apiKey}`;

    let parts = [];
    if (typeof user === 'string') {
        parts = [{ text: user }];
    } else if (user._multimodal) {
        parts = user.gemini;
    } else if (Array.isArray(user)) {
        parts = user;
    }

    const body = {
        contents: [{ role: 'user', parts: parts }],
        generationConfig: { temperature: CFG.TEMPERATURE, maxOutputTokens: 2000 }
    };
    if (system) {
         body.systemInstruction = { parts: [{ text: system }] };
    }

    let delay = 1000;
    for (let i = 0; i < 5; i++) {
        try {
            const res = await fetch(url, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(body) });
            if (!res.ok) {
                if (res.status >= 400 && res.status < 500 && res.status !== 429) {
                    const errData = await res.text();
                    throw new Error(`API Error ${res.status}: ${errData}`);
                }
                throw new Error(`HTTP ${res.status}`);
            }
            const data = await res.json();
            const text = data.candidates?.[0]?.content?.parts?.[0]?.text;
            if (text) return text;
            throw new Error('Invalid Gemini Canvas response structure');
        } catch (err) {
            if (i === 4 || err.message.includes('API Error 400') || err.message.includes('API Error 403')) throw err;
            await new Promise(r => setTimeout(r, delay));
            delay *= 2;
        }
    }
}

    const url = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-09-2025:generateContent?key=${apiKey}`;
    const body = {

            { text: 'Transcribe this audio verbatim. Output ONLY the spoken words. If no clear speech, output nothing.' },
        ]}],
        generationConfig: { temperature: 0 }
    };

    let delay = 1000;
    for(let i = 0; i < 5; i++){
        try {
            const res = await fetch(url, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(body) });
            if(!res.ok) throw new Error(`Audio API HTTP ${res.status}`);

        } catch(e) {
            if (i === 4) throw e;

        }
    }

function migrateTagSchema(item) {
    if (!item) return item;
    if (!Array.isArray(item.tags))    item.tags     = [];
    if (item.archived === undefined)   item.archived = false;
    if (item.rating   === undefined)   item.rating   = 0;
    return item;
}

function migrateLearnSchema() {
    if (!learningExamples.length || learningExamples[0].id) return;
    createAutoBackup();
    learningExamples = learningExamples.map(ex => ({
        id: crypto.randomUUID(),
        original: ex.original || '',
        edited: ex.edited || '',
        applianceType: '',
        timestamp: ex.timestamp || Date.now(),
        rating: 3,
        charCount: (ex.edited || '').length,
        diff: null
    }));
    saveState();
}

function pruneExamples() {
    const before = learningExamples.length;
    learningExamples = learningExamples.filter(ex => decayWeight(ex) >= 0.05);
    if (learningExamples.length < before) {
        saveState();
    }
}

function migrateAllTagSchemas() {
    if (catalogData.length && catalogData[0].archived !== undefined) return;
    if (!catalogData.length && !brainData.length && !savedLogs.length) return;
    createAutoBackup();
    catalogData.forEach(migrateTagSchema);
    brainData.forEach(migrateTagSchema);
    savedLogs.forEach(l => { if (l.structured) migrateTagSchema(l.structured); });
    saveState();
    rebuildTagRegistry();
}

function rebuildTagRegistry() {
    const all = new Set();
    const addTags = (arr) => arr.forEach(item => (item.tags || []).forEach(t => all.add(t.toLowerCase())));
    addTags(catalogData);
    addTags(brainData);
    savedLogs.forEach(l => (l.structured?.tags || []).forEach(t => all.add(t.toLowerCase())));
    lsSet(LS.TAGS, [...all]);
    return all;
}

function detectApplianceType(model) {
    if (!model) return '';
    const m = model.toUpperCase().trim();
    if (/^(WTW|MVW|CAW|NTW|WFW|MFW|GFW|CFW|HFW|WOW)/.test(m)) return 'Front/Top Load Washer';
    if (/^(WED|MED|GEW|CED|NED|WGD|MGD|GGD|CGD|NGD)/.test(m)) return 'Dryer';
    if (/^(W[RQ]F|WRS|WRB|WRX|MFI|MFI|MFI|WRT|WRC|MRT|MSF|MSS|MSC|GSF|GSC|GSS|GS[FCS])/.test(m)) return 'Refrigerator';
    if (/^(WFE|WFG|WEE|WEG|MFE|MFG|MEE|MEG|GFE|GFG|WOS|WOC|MOS|MOC)/.test(m)) return 'Range/Oven';
    if (/^(WDF|MDB|WDT|WDTA|GU[0-9]|GUD)/.test(m)) return 'Dishwasher';
    if (/^(WMH|MMV|GMH|IMH|UMV|YMH)/.test(m)) return 'Microwave';
    if (/^(WZF|MZF)/.test(m)) return 'Freezer';
    return '';
}

function compressTranscript(text) {
    if (!text) return text;
    const words = text.trim().split(/\s+/).length;
    if (words < 200) return text;
    return text
        .replace(/\b(?:uh+|um+|umm+|uhh+)\b[,]?\s*/gi, ' ')
        .replace(/\byou know\b[,]?\s*/gi, ' ')
        .trim();
}

async function startMic(useLabel) {
    const Speech = window.SpeechRecognition || window.webkitSpeechRecognition;
    if (!Speech) return;
    try {
        const stream = await navigator.mediaDevices.getUserMedia({ audio: true });
        stream.getTracks().forEach(t => t.stop());
    } catch (err) {
        isRecording = false; stopVoice(); toast('❌ Mic denied'); return;
    }
    micRecognition = new Speech();
    micRecognition.continuous = true; micRecognition.interimResults = true;
    micRecognition.onresult = (ev) => {
        if (!isRecording) return;
        let interim = '', final = '';
        for (let i = ev.resultIndex; i < ev.results.length; i++) {
            if (ev.results[i].isFinal) final += ev.results[i][0].transcript;
            else interim += ev.results[i][0].transcript;
        }
        const ib = $('interimBar'); if(ib) ib.textContent = interim ? `🎙️ ${interim}` : '…';
        if (final) appendTranscript(final, useLabel ? 'TECH' : null);
    };
    micRecognition.onstart = () => { micActive = true; const dt = $('dotTech'); if(dt) dt.classList.add('on'); };
    micRecognition.onend = () => { 
        if (isRecording && micActive) {
            try { micRecognition.start(); } catch {}
        } else { 
            micActive = false; 
            const dt = $('dotTech'); if(dt) dt.classList.remove('on'); 
        } 
    };
    micRecognition.start();
}

function stopMic() { 
    micActive = false; 
    if (micRecognition) {
        micRecognition.onend = null; 
        micRecognition.onresult = null;
        try { micRecognition.stop(); micRecognition.abort(); } catch {}
        micRecognition = null;
    }
}

async function startSysAudio() {
    try {
        const ds = await navigator.mediaDevices.getDisplayMedia({ video: { width: 1 }, audio: true });
        ds.getVideoTracks().forEach(t => t.stop());
        const at = ds.getAudioTracks(); if (!at.length) return false;
        sysStream = new MediaStream(at);
        sysRecorder = new MediaRecorder(sysStream, { mimeType: 'audio/webm;codecs=opus' });
        sysRecorder.ondataavailable = async (e) => {
            if (e.data.size < 500) return;
            setCxStatus('transcribing…');
            try {
                const b64 = await blobToBase64(e.data);
                const text = await transcribeAudio(b64);
                if (text?.trim()) appendTranscript(text.trim(), 'CX');
            } catch {} finally { setCxStatus(''); }
        };
        sysRecorder.start(6000); sysActive = true;
        const dc = $('dotCx'); if(dc) { dc.style.background = 'var(--cx)'; dc.classList.add('on'); }
        at[0].onended = () => stopSysAudio();
        return true;
    } catch { return false; }
}

async function regenerateSection(sectionKey) {
    if (!lastSections[sectionKey]?.parseable) return;
    const inputs = gatherNoteInputs();
    const current = ($('finalOutput')?.value || '').replace(/\n\n— Generated:.*$/m, '').trim();
    const btn = document.querySelector(`[data-regen="${sectionKey}"]`);
    if(!btn) return;
    setBusy(btn, '…');
    const system = buildSystemPrompt() + '\n\nOutput ONLY the plain text bullets for the requested section. No headers.';
    const user = buildUserInput({ ...inputs, transcript: compressTranscript(inputs.transcript) }) + `\n\nTASK: ${REGEN_PROMPTS[sectionKey]}`;
    try {
        const cleaned = (await callAI(system, user)).trim();
        const pattern = SECTION_PATTERNS[sectionKey];
        const m = current.match(pattern);
        if (m) {
            const updated = current.substring(0, m.index) + m[0].replace(m[1], '\n' + cleaned + '\n') + current.substring(m.index + m[0].length);
            const stamp = `\n\n— Generated: ${inputs.model || 'No model'} | ${new Date().toLocaleTimeString()} | ${ENV} | ${CFG.VERSION}`;
            $('finalOutput').value = updated + stamp;
            lastSections = parseOutputSections(updated);
        }
    } catch {} finally { setReady(btn, '↻ ' + REGEN_SECTIONS.find(s => s.key === sectionKey).label); }
}

function extractStructured(output) {
    if (!output) return null;
    const mm = output.match(/Model:\s*([A-Z0-9\-]{5,17})/i);
    const model = mm ? mm[1].trim().toUpperCase() : '';
    const cSec = output.match(/^Complaint:\s*\n([\s\S]*?)(?=^Prior Repairs:|^Why|^Before|^During|^Interpretation|^Resolution|^Case:|$)/m);
    const syms = cSec ? (cSec[1].match(/^\s*[-•*]?\s*([^\n]+)/gm) || []).map(b => b.replace(/^\s*[-•*]?\s*/, '').trim()) : [];
    const dSec = output.match(/^During call:\s*\n([\s\S]*?)(?=^Interpretation:|^Resolution:|^Case:|$)/m);
    const comps = dSec ? (dSec[1].match(/^\s*[-•*]?\s*([^\n]+)/gm) || []).map(b => ({ name: b.replace(/^\s*[-•*]?\s*/, '').trim(), status: 'unknown' })) : [];
    const iSec = output.match(/^Interpretation:\s*\n([\s\S]*?)(?=^Resolution:|^Case:|$)/m);
    const cause = iSec ? iSec[1].trim() : '';
    return { model, symptomTags: syms, componentsTested: comps, rootCause: cause, outcomeTag: 'unknown' };
}

function buildSheetsPayload() {

    return {
        date: new Date().toLocaleString(),

        concern: inputs.concern,
        fullNote: rawOut,
        callSec: Math.floor(callMs / 1000),
        outcomeTag: structured?.outcomeTag || '',
        env: ENV, ttsVersion: CFG.VERSION,
    };
}

function filterSort(cases, filter, sort, search, showArchived) {
    let f = cases.filter(c => showArchived ? true : !c.archived);
    if (filter !== 'all') f = f.filter(c => c.type === filter);
    if (search) { const lo = search.toLowerCase(); f = f.filter(c => c.model.toLowerCase().includes(lo) || c.content.toLowerCase().includes(lo)); }
    if (sort === 'recent') f.sort((a, b) => b.timestamp - a.timestamp);
    return f;
}

        const el = $(`sec-${k}`);

            const n = el.value.split('\n').filter(l=>l.trim()).length;
            const countEl = $(`cnt-${k}`);
            if(countEl) countEl.textContent = n || '';

    });
}
